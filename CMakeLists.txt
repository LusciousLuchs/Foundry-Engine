cmake_minimum_required(VERSION 3.24)
project(Foundry VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(FOUNDRY_BUILD_TESTS "Build tests" ON)
option(FOUNDRY_BACKEND_VULKAN "Build the Vulkan RHI backend" OFF)
option(FOUNDRY_BACKEND_OPENGL "Build the OpenGL RHI backend" OFF)

add_library(foundry_core INTERFACE)
target_include_directories(foundry_core INTERFACE engine)
target_compile_features(foundry_core INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(foundry_core INTERFACE /W4)
else()
  target_compile_options(foundry_core INTERFACE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endif()

add_library(foundry_rhi INTERFACE)
target_link_libraries(foundry_rhi INTERFACE foundry_core)

if(WIN32)
  add_library(foundry_platform STATIC engine/platform/Window_win32.cpp)
  target_link_libraries(foundry_platform PUBLIC foundry_core)

  # Khronos Vulkan headers are vendored (Apache-2.0 / MIT); the loader is vulkan-1.dll at runtime.
  set(VULKAN_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/vulkan-headers/include)
  if(EXISTS ${VULKAN_INCLUDE_DIR}/vulkan/vulkan.h)
    set(FOUNDRY_BACKEND_VULKAN ON)
    add_library(foundry_vulkan STATIC engine/rhi/vulkan/VkContext.cpp)
    target_include_directories(foundry_vulkan PUBLIC ${VULKAN_INCLUDE_DIR})
    target_link_libraries(foundry_vulkan PUBLIC foundry_rhi)
    add_executable(triangle examples/triangle.cpp)
    target_link_libraries(triangle PRIVATE foundry_platform foundry_vulkan)
    target_link_options(triangle PRIVATE -mwindows)
  else()
    message(STATUS "Vulkan headers not found: Vulkan backend disabled")
  endif()
endif()

if(FOUNDRY_BUILD_TESTS)
  enable_testing()
  add_executable(foundry_tests tests/main.cpp)
  target_link_libraries(foundry_tests PRIVATE foundry_core foundry_rhi)
  add_test(NAME foundry_tests COMMAND foundry_tests)

  # Optional: verify compat math against the real raymath.h of the game being ported.
  set(FOUNDRY_RAYLIB_DIR "" CACHE PATH "Directory containing raylib.h and raymath.h")
  if(FOUNDRY_RAYLIB_DIR)
    add_executable(raymath_diff tests/raymath_diff.cpp)
    target_include_directories(raymath_diff PRIVATE ${FOUNDRY_RAYLIB_DIR})
    target_link_libraries(raymath_diff PRIVATE foundry_core)
    if(NOT MSVC)
      target_compile_options(raymath_diff PRIVATE -Wno-missing-field-initializers)
    endif()
    add_test(NAME raymath_diff COMMAND raymath_diff)
  endif()
endif()
